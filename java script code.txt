const $ = (selector, root = document) => root.querySelector(selector);
const $$ = (selector, root = document) =>
  Array.from(root.querySelectorAll(selector));

// =========================================================
// MOBILE NAVIGATION
// =========================================================

const menuToggle = $(".menu-toggle");
const mainNavigation = $("#main-navigation");

function closeMobileMenu() {
  if (!menuToggle || !mainNavigation) return;

  menuToggle.setAttribute("aria-expanded", "false");
  menuToggle.setAttribute("aria-label", "Apri il menu");
  mainNavigation.classList.remove("is-open");
  document.body.classList.remove("menu-open");
}

function openMobileMenu() {
  if (!menuToggle || !mainNavigation) return;

  menuToggle.setAttribute("aria-expanded", "true");
  menuToggle.setAttribute("aria-label", "Chiudi il menu");
  mainNavigation.classList.add("is-open");
  document.body.classList.add("menu-open");
}

if (menuToggle && mainNavigation) {
  menuToggle.addEventListener("click", () => {
    const isExpanded =
      menuToggle.getAttribute("aria-expanded") === "true";

    if (isExpanded) {
      closeMobileMenu();
    } else {
      openMobileMenu();
    }
  });

  // Close the mobile menu after selecting a navigation destination.
  $$("a", mainNavigation).forEach((link) => {
    link.addEventListener("click", closeMobileMenu);
  });

  // Close the menu with Escape and return focus to the toggle.
  document.addEventListener("keydown", (event) => {
    if (
      event.key === "Escape" &&
      menuToggle.getAttribute("aria-expanded") === "true"
    ) {
      closeMobileMenu();
      menuToggle.focus();
    }
  });

  // Restore the desktop navigation when the viewport expands.
  window.addEventListener("resize", () => {
    if (window.innerWidth > 760) {
      closeMobileMenu();
    }
  });
}

// =========================================================
// ACTIVE NAVIGATION HIGHLIGHT
// =========================================================

const navigationLinks = $$(".main-navigation a[href^='#']");
const observedSections = $$(
  "#salone, #servizi, #recensioni, #contatti"
);

if ("IntersectionObserver" in window && observedSections.length) {
  const sectionObserver = new IntersectionObserver(
    (entries) => {
      // Select the visible section nearest the top of the viewport.
      const visibleEntries = entries
        .filter((entry) => entry.isIntersecting)
        .sort(
          (a, b) =>
            a.boundingClientRect.top - b.boundingClientRect.top
        );

      if (!visibleEntries.length) return;

      const activeId = visibleEntries[0].target.id;

      navigationLinks.forEach((link) => {
        if (link.getAttribute("href") === `#${activeId}`) {
          link.setAttribute("aria-current", "location");
        } else {
          link.removeAttribute("aria-current");
        }
      });
    },
    {
      rootMargin: "-15% 0px -65% 0px",
      threshold: 0
    }
  );

  observedSections.forEach((section) => {
    sectionObserver.observe(section);
  });
}

// DATE INPUT: PREVENT PAST DATES
// =========================================================

const dateInput = $("#preferred-date");

function getLocalDateString(date = new Date()) {
  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, "0");
  const day = String(date.getDate()).padStart(2, "0");

  return `${year}-${month}-${day}`;
}

if (dateInput) {
  dateInput.min = getLocalDateString();
}

// IMPORTANT:
// The static form below does NOT send email, save data or create bookings.
// Before launch, connect a form provider, serverless endpoint, email service,
// confirmed WhatsApp number or booking platform.
//
// When a confirmed WhatsApp number is configured, valid form submissions
// open a pre-filled WhatsApp message. The customer still has to press Send,
// and the salon must confirm availability.
//
// Never tell customers a request has been delivered unless a real service
// has confirmed that it was received.

const appointmentForm = $("#appointment-form");
const formStatus = $("#form-status");

const formFields = {
  name: $("#client-name"),
  phone: $("#client-phone"),
  date: $("#preferred-date"),
  time: $("#preferred-time"),
  service: $("#requested-service"),
  notes: $("#client-notes")
};

const validationRules = [
  {
    key: "name",
    errorId: "#name-error",
    message: "Inserisci il tuo nome e cognome.",
    isValid: (value) => value.trim().length >= 2
  },
  {
    key: "phone",
    errorId: "#phone-error",
    message: "Inserisci un numero di telefono valido.",
    isValid: (value) => {
      const digits = value.replace(/\D/g, "");
      return digits.length >= 7 && digits.length <= 15;
    }
  },
  {
    key: "date",
    errorId: "#date-error",
    message: "Scegli una data valida, da oggi in poi.",
    isValid: (value) =>
      /^\d{4}-\d{2}-\d{2}$/.test(value) &&
      value >= getLocalDateString()
  },
  {
    key: "time",
    errorId: "#time-error",
    message: "Seleziona un orario preferito.",
    isValid: (value) => value.trim() !== ""
  },
  {
    key: "service",
    errorId: "#service-error",
    message: "Seleziona il servizio desiderato.",
    isValid: (value) => value.trim() !== ""
  }
];

function setFieldError(field, errorElement, message) {
  if (!field || !errorElement) return;

  errorElement.textContent = message;

  if (message) {
    field.classList.add("is-invalid");
    field.setAttribute("aria-invalid", "true");
  } else {
    field.classList.remove("is-invalid");
    field.removeAttribute("aria-invalid");
  }
}

function clearFormStatus() {
  if (!formStatus) return;

  formStatus.textContent = "";
  formStatus.classList.remove("is-error", "is-success");
}

function showFormStatus(message, type = "error") {
  if (!formStatus) return;

  formStatus.textContent = message;
  formStatus.classList.remove("is-error", "is-success");
  formStatus.classList.add(
    type === "success" ? "is-success" : "is-error"
  );
}

function validateAppointmentForm() {
  let isValid = true;
  let firstInvalidField = null;

  validationRules.forEach((rule) => {
    const field = formFields[rule.key];
    const errorElement = $(rule.errorId);
    const valid = field && rule.isValid(field.value);

    setFieldError(
      field,
      errorElement,
      valid ? "" : rule.message
    );

    if (!valid) {
      isValid = false;

      if (!firstInvalidField) {
        firstInvalidField = field;
      }
    }
  });

  if (firstInvalidField) {
    firstInvalidField.focus();
  }

  return isValid;
}

// Clear a field's displayed error when the customer corrects it.
validationRules.forEach((rule) => {
  const field = formFields[rule.key];

  if (!field) return;

  const eventName =
    field.tagName === "SELECT" ? "change" : "input";

  field.addEventListener(eventName, () => {
    const errorElement = $(rule.errorId);

    if (rule.isValid(field.value)) {
      setFieldError(field, errorElement, "");
    }

    clearFormStatus();
  });
});

if (appointmentForm) {
  appointmentForm.addEventListener("submit", (event) => {
    event.preventDefault();
    clearFormStatus();

    if (!validateAppointmentForm()) {
      showFormStatus(
        "Controlla i campi evidenziati e riprova."
      );
      return;
    }

    const bookingRequest = {
      name: formFields.name.value.trim(),
      phone: formFields.phone.value.trim(),
      date: formFields.date.value,
      time: formFields.time.value,
      service: formFields.service.value,
      notes: formFields.notes.value.trim()
    };

    // Path A: a WhatsApp number has been confirmed by the owner.
    if (SITE_CONFIG.whatsappNumber) {
      const message = [
        "Ciao Katia! Vorrei richiedere un appuntamento.",
        "",
        `Nome: ${bookingRequest.name}`,
        `Telefono: ${bookingRequest.phone}`,
        `Data preferita: ${bookingRequest.date}`,
        `Orario preferito: ${bookingRequest.time}`,
        `Servizio: ${bookingRequest.service}`,
        bookingRequest.notes
          ? `Note: ${bookingRequest.notes}`
          : "",
        "",
        "So che l'appuntamento deve essere confermato dal salone."
      ]
        .filter(Boolean)
        .join("\n");

      const whatsappUrl =
        `https://wa.me/${SITE_CONFIG.whatsappNumber}` +
        `?text=${encodeURIComponent(message)}`;

      // Opens a draft only. The user must review and send it.
      window.open(whatsappUrl, "_blank", "noopener,noreferrer");

      showFormStatus(
        "Si è aperta una bozza del messaggio WhatsApp, se il browser lo consente. Controlla il messaggio e premi Invia. L'appuntamento non è ancora confermato.",
        "success"
      );

      return;
    }

    // Path B: no real form submission service has been configured.
    // Do not store, send or falsely claim to have submitted the data.
    showFormStatus(
      "Questa richiesta non è stata inviata né salvata. Per richiedere un appuntamento, chiama il salone al 035 529 2147. Il modulo dovrà essere collegato a un servizio di invio prima di poter essere utilizzato online."
    );
  });
}
